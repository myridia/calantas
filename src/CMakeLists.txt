nap_desktop_app(calantas "" NRC_NONE)
target_include_directories(calantas PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(calantas PRIVATE
  CALANTAS_TELEGRAM_APP_API_ID="${CALANTAS_TELEGRAM_APP_API_ID}"
  CALANTAS_TELEGRAM_APP_API_HASH="${CALANTAS_TELEGRAM_APP_API_HASH}")

if(CALANTAS_XMPP_TARGET)
  target_compile_definitions(calantas PRIVATE CALANTAS_HAVE_XMPP=1)
  target_link_libraries(calantas ${CALANTAS_XMPP_TARGET})
endif()

if(CALANTAS_TELEGRAM_TARGET)
  target_compile_definitions(calantas PRIVATE CALANTAS_HAVE_TELEGRAM=1)
  target_link_libraries(calantas ${CALANTAS_TELEGRAM_TARGET})
endif()

set(CALANTAS_BACKEND_RPATH ${CALANTAS_XMPP_RPATH} ${CALANTAS_TELEGRAM_RPATH})
if(CALANTAS_BACKEND_RPATH)
  list(REMOVE_DUPLICATES CALANTAS_BACKEND_RPATH)
  set_property(TARGET calantas APPEND PROPERTY BUILD_RPATH ${CALANTAS_BACKEND_RPATH})
  set_property(TARGET calantas APPEND PROPERTY INSTALL_RPATH ${CALANTAS_BACKEND_RPATH})
endif()
