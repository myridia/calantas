cmake_minimum_required(VERSION 3.10)
project(calantas)

option(CALANTAS_ENABLE_XMPP "Build the XMPP backend when libstrophe is available" ON)
option(CALANTAS_REQUIRE_XMPP "Fail configuration when libstrophe is missing" OFF)
option(CALANTAS_ENABLE_TELEGRAM "Build the Telegram backend when TDLib is available" ON)
option(CALANTAS_REQUIRE_TELEGRAM "Fail configuration when TDLib is missing" OFF)

if(DEFINED ENV{NAPPGUI_DIR})
  set(nappgui_DIR "$ENV{NAPPGUI_DIR}/cmake" CACHE PATH "NAppGUI cmake directory")
elseif(WIN32)
  set(nappgui_DIR "C:/nappgui/cmake" CACHE PATH "NAppGUI cmake directory")
else()
  set(nappgui_DIR "/usr/local/nappgui/cmake" CACHE PATH "NAppGUI cmake directory")
endif()

if(NOT EXISTS "${nappgui_DIR}/nappgui-config.cmake")
  message(FATAL_ERROR "NAppGUI config not found at ${nappgui_DIR}/nappgui-config.cmake. Run ask.sh task 2 or set NAPPGUI_DIR to the SDK prefix.")
endif()

find_package(nappgui REQUIRED)
include("${NAPPGUI_ROOT_PATH}/prj/NAppProject.cmake")
include("${NAPPGUI_ROOT_PATH}/prj/NAppCompilers.cmake")

set(CALANTAS_XMPP_TARGET "")
set(CALANTAS_XMPP_RPATH "")

if(CALANTAS_ENABLE_XMPP)
  find_package(PkgConfig QUIET)
  if(PkgConfig_FOUND)
    pkg_check_modules(PC_STROPHE QUIET IMPORTED_TARGET libstrophe)
  endif()

  if(TARGET PkgConfig::PC_STROPHE)
    set(CALANTAS_XMPP_TARGET PkgConfig::PC_STROPHE)
    set(CALANTAS_XMPP_RPATH "${PC_STROPHE_LIBRARY_DIRS}")
  else()
    find_path(STROPHE_INCLUDE_DIR NAMES strophe.h)
    find_library(STROPHE_LIBRARY NAMES strophe libstrophe)
    if(STROPHE_INCLUDE_DIR AND STROPHE_LIBRARY)
      set(STROPHE_EXTRA_LIBS "")
      foreach(STROPHE_DEP expat ssl crypto z)
        find_library(STROPHE_DEP_${STROPHE_DEP} NAMES ${STROPHE_DEP})
        if(STROPHE_DEP_${STROPHE_DEP})
          list(APPEND STROPHE_EXTRA_LIBS "${STROPHE_DEP_${STROPHE_DEP}}")
        endif()
      endforeach()
      add_library(calantas_libstrophe UNKNOWN IMPORTED)
      set_target_properties(calantas_libstrophe PROPERTIES
        IMPORTED_LOCATION "${STROPHE_LIBRARY}"
        INTERFACE_INCLUDE_DIRECTORIES "${STROPHE_INCLUDE_DIR}"
        INTERFACE_LINK_LIBRARIES "${STROPHE_EXTRA_LIBS}")
      get_filename_component(CALANTAS_XMPP_RPATH "${STROPHE_LIBRARY}" DIRECTORY)
      set(CALANTAS_XMPP_TARGET calantas_libstrophe)
    endif()
  endif()
endif()

if(CALANTAS_XMPP_TARGET)
  message(STATUS "XMPP backend: enabled (${CALANTAS_XMPP_TARGET})")
elseif(NOT CALANTAS_ENABLE_XMPP)
  message(STATUS "XMPP backend: disabled by CALANTAS_ENABLE_XMPP=OFF")
  if(CALANTAS_REQUIRE_XMPP)
    message(FATAL_ERROR "CALANTAS_REQUIRE_XMPP is on but CALANTAS_ENABLE_XMPP is off")
  endif()
else()
  message(STATUS "XMPP backend: disabled (libstrophe not found)")
  if(CALANTAS_REQUIRE_XMPP)
    message(FATAL_ERROR "libstrophe not found. Install it, pass -DCMAKE_PREFIX_PATH=<libstrophe prefix>, or turn off CALANTAS_REQUIRE_XMPP.")
  endif()
endif()

set(CALANTAS_TELEGRAM_TARGET "")
set(CALANTAS_TELEGRAM_RPATH "")

if(CALANTAS_ENABLE_TELEGRAM)
  find_package(PkgConfig QUIET)
  if(PkgConfig_FOUND)
    pkg_check_modules(PC_TDJSON QUIET IMPORTED_TARGET tdjson)
  endif()

  if(TARGET PkgConfig::PC_TDJSON)
    set(CALANTAS_TELEGRAM_TARGET PkgConfig::PC_TDJSON)
    set(CALANTAS_TELEGRAM_RPATH "${PC_TDJSON_LIBRARY_DIRS}")
  else()
    find_library(TDJSON_LIBRARY NAMES tdjson libtdjson)
    if(TDJSON_LIBRARY)
      set(TDJSON_EXTRA_LIBS "")
      if(UNIX AND NOT APPLE)
        foreach(TDJSON_DEP pthread dl)
          find_library(TDJSON_DEP_${TDJSON_DEP} NAMES ${TDJSON_DEP})
          if(TDJSON_DEP_${TDJSON_DEP})
            list(APPEND TDJSON_EXTRA_LIBS "${TDJSON_DEP_${TDJSON_DEP}}")
          endif()
        endforeach()
        find_library(TDJSON_DEP_CXX NAMES c++)
        if(TDJSON_DEP_CXX)
          list(APPEND TDJSON_EXTRA_LIBS "${TDJSON_DEP_CXX}")
        endif()
      endif()
      add_library(calantas_tdjson UNKNOWN IMPORTED)
      set_target_properties(calantas_tdjson PROPERTIES
        IMPORTED_LOCATION "${TDJSON_LIBRARY}"
        INTERFACE_LINK_LIBRARIES "${TDJSON_EXTRA_LIBS}")
      get_filename_component(CALANTAS_TELEGRAM_RPATH "${TDJSON_LIBRARY}" DIRECTORY)
      set(CALANTAS_TELEGRAM_TARGET calantas_tdjson)
    endif()
  endif()
endif()

if(CALANTAS_TELEGRAM_TARGET)
  message(STATUS "Telegram backend: enabled (${CALANTAS_TELEGRAM_TARGET})")
elseif(NOT CALANTAS_ENABLE_TELEGRAM)
  message(STATUS "Telegram backend: disabled by CALANTAS_ENABLE_TELEGRAM=OFF")
  if(CALANTAS_REQUIRE_TELEGRAM)
    message(FATAL_ERROR "CALANTAS_REQUIRE_TELEGRAM is on but CALANTAS_ENABLE_TELEGRAM is off")
  endif()
else()
  message(STATUS "Telegram backend: disabled (tdjson not found)")
  if(CALANTAS_REQUIRE_TELEGRAM)
    message(FATAL_ERROR "tdjson not found. Install TDLib, pass -DCMAKE_PREFIX_PATH=<TDLib prefix>, or turn off CALANTAS_REQUIRE_TELEGRAM.")
  endif()
endif()

message (STATUS "---------------------------------------------")
message (STATUS "calantas messenger")
message (STATUS "info@myridia.com")
message (STATUS "GNU GENERAL PUBLIC License")
message(STATUS ${CMAKE_CXX_COMPILER_ID})
message (STATUS "---------------------------------------------")
nap_config_compiler()
nap_project_desktop_app(calantas src)
